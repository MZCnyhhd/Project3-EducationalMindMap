滚轮缩放 · 拖拽平移
中心=公式 · 子节点=每个符号 · 橙色=整体意义
viewport.addEventListener('wheel',e=>{ e.preventDefault(); const delta=-e.deltaY*0.0015; const ns=Math.min(2.5,Math.max(0.3,scale*(1+delta))); const rect=viewport.getBoundingClientRect(); const cx=e.clientX-rect.left, cy=e.clientY-rect.top; tx = cx-(cx-tx)*(ns/scale); ty=cy-(cy-ty)*(ns/scale); scale=ns; applyTransform(); },{passive:false}); let dragging=false, sx,sy, stx,sty; viewport.addEventListener('mousedown',e=>{ dragging=true; sx=e.clientX; sy=e.clientY; stx=tx; sty=ty; viewport.classList.add('grabbing'); }); window.addEventListener('mousemove',e=>{ if(!dragging)return; tx=stx+(e.clientX-sx); ty=sty+(e.clientY-sy); applyTransform(); }); window.addEventListener('mouseup',()=>{ dragging=false; viewport.classList.remove('grabbing'); }); document.getElementById('zoomIn').onclick=()=>{ scale=Math.min(2.5,scale*1.2); applyTransform(); }; document.getElementById('zoomOut').onclick=()=>{ scale=Math.max(0.3,scale*0.8); applyTransform(); }; document.getElementById('reset').onclick=()=>{ tx=60; ty=40; scale=1; applyTransform(); };
